cmake_minimum_required(VERSION 3.20)
project(gscvm VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

option(GSCVM_BUILD_UPSTREAM_CLI "Build the upstream gsc-tool executable" ON)
option(GSCVM_BUILD_UPSTREAM_TESTS "Build the upstream test suite" ON)
option(GSCVM_BUILD_TESTS "Build gscvm tests" ON)
option(GSCVM_BUILD_WASM "Build the browser WebAssembly playground module" OFF)

if(GSCVM_BUILD_WASM)
    if(NOT EMSCRIPTEN)
        message(FATAL_ERROR "GSCVM_BUILD_WASM requires an Emscripten toolchain (use emcmake cmake ...)")
    endif()

    # The browser target is a standalone library; the native CLI and the
    # upstream test corpus are not useful inside the WASM build.
    set(GSCVM_BUILD_UPSTREAM_CLI OFF CACHE BOOL "Build the upstream gsc-tool executable" FORCE)
    set(GSCVM_BUILD_UPSTREAM_TESTS OFF CACHE BOOL "Build the upstream test suite" FORCE)
    set(GSCVM_BUILD_TESTS OFF CACHE BOOL "Build gscvm tests" FORCE)

    add_compile_options(-fexceptions)
    add_link_options(-fexceptions)
endif()

set(UPSTREAM_DIR "${CMAKE_CURRENT_SOURCE_DIR}/upstream")
set(GENERATED_INCLUDE_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated/include")
file(MAKE_DIRECTORY "${GENERATED_INCLUDE_DIR}/xsk")

# premake normally generates this file. The VM does not depend on it; it is only
# needed by the upstream CLI.
file(WRITE "${GENERATED_INCLUDE_DIR}/xsk/version.hpp"
"// Generated by the gscvm CMake build - do not edit\n\n"
"#define XSK_VERSION_MAJOR 0\n"
"#define XSK_VERSION_MINOR 1\n"
"#define XSK_VERSION_PATCH 0\n"
"#define XSK_VERSION_BUILD 0\n"
"#define XSK_VERSION_BRANCH \"gscvm\"\n"
"#define XSK_VERSION_STR \"0.1.0-gscvm\"\n")

set(UPSTREAM_INCLUDES
    "${UPSTREAM_DIR}/include"
    "${GENERATED_INCLUDE_DIR}"
)

# ---------------------------------------------------------------------------
# Vendored zlib (upstream dependency)
# ---------------------------------------------------------------------------
file(GLOB XSK_ZLIB_SOURCES CONFIGURE_DEPENDS "${UPSTREAM_DIR}/deps/zlib/*.c")
add_library(xsk_zlib STATIC ${XSK_ZLIB_SOURCES})
target_include_directories(xsk_zlib PUBLIC "${UPSTREAM_DIR}/deps/zlib")
if(NOT WIN32)
    target_compile_definitions(xsk_zlib PRIVATE HAVE_UNISTD_H)
endif()
set_target_properties(xsk_zlib PROPERTIES C_STANDARD 99)

# ---------------------------------------------------------------------------
# Upstream libraries, reused as-is
# ---------------------------------------------------------------------------
file(GLOB_RECURSE XSK_UTILS_SOURCES CONFIGURE_DEPENDS "${UPSTREAM_DIR}/src/utils/*.cpp")
add_library(xsk_utils STATIC ${XSK_UTILS_SOURCES})
target_include_directories(xsk_utils PUBLIC ${UPSTREAM_INCLUDES})
target_link_libraries(xsk_utils PUBLIC xsk_zlib)

file(GLOB_RECURSE XSK_ARC_SOURCES CONFIGURE_DEPENDS "${UPSTREAM_DIR}/src/arc/*.cpp")

# gscvm needs OP_ProfileStart to be treated as an operand-less marker. Keep the
# upstream checkout pristine and generate a patched copy of the one affected
# translation unit instead of mutating the submodule.
set(XSK_DISASSEMBLER_SOURCE "${UPSTREAM_DIR}/src/arc/disassembler.cpp")
set(XSK_DISASSEMBLER_MARKER
"        case opcode::OP_NextArrayKey:
        // case opcode::OP_ProfileStart:
        case opcode::OP_ProfileStop:")
file(READ "${XSK_DISASSEMBLER_SOURCE}" XSK_DISASSEMBLER_TEXT)
string(FIND "${XSK_DISASSEMBLER_TEXT}" "${XSK_DISASSEMBLER_MARKER}" XSK_DISASSEMBLER_MARKER_POS)
if(NOT XSK_DISASSEMBLER_MARKER_POS EQUAL -1)
    string(REPLACE "${XSK_DISASSEMBLER_MARKER}"
"        case opcode::OP_NextArrayKey:
        // gscvm: OP_ProfileStart has no operand in the T6 encoding; T7's wide form
        // is handled by opcode_size() and this branch only decides what to read.
        case opcode::OP_ProfileStart:
        case opcode::OP_ProfileStop:"
        XSK_DISASSEMBLER_TEXT "${XSK_DISASSEMBLER_TEXT}")
    set(XSK_DISASSEMBLER_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated/upstream/src/arc/disassembler.cpp")
    file(WRITE "${XSK_DISASSEMBLER_GENERATED}" "${XSK_DISASSEMBLER_TEXT}")
    list(REMOVE_ITEM XSK_ARC_SOURCES "${XSK_DISASSEMBLER_SOURCE}")
    list(APPEND XSK_ARC_SOURCES "${XSK_DISASSEMBLER_GENERATED}")
endif()

add_library(xsk_arc STATIC ${XSK_ARC_SOURCES})
target_include_directories(xsk_arc PUBLIC ${UPSTREAM_INCLUDES})
target_link_libraries(xsk_arc PUBLIC xsk_utils)
# GCC treats xsk::arc::instruction's `opcode opcode` member as an error under
# the C++20 "changes meaning" rule. Permissive mode keeps the upstream source
# untouched and is only used for the vendored compatibility library.
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(xsk_arc PRIVATE -fpermissive)
endif()
if(MSVC)
    target_compile_options(xsk_arc PRIVATE /bigobj)
endif()

if(NOT GSCVM_BUILD_WASM)
    file(GLOB_RECURSE XSK_GSC_SOURCES CONFIGURE_DEPENDS "${UPSTREAM_DIR}/src/gsc/*.cpp")
    add_library(xsk_gsc STATIC ${XSK_GSC_SOURCES})
    target_include_directories(xsk_gsc PUBLIC ${UPSTREAM_INCLUDES})
    target_link_libraries(xsk_gsc PUBLIC xsk_utils)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_compile_options(xsk_gsc PRIVATE -fpermissive)
    endif()
    if(MSVC)
        target_compile_options(xsk_gsc PRIVATE /bigobj)
    endif()
endif()

# ---------------------------------------------------------------------------
# gscvm library
# ---------------------------------------------------------------------------
file(GLOB_RECURSE GSCVM_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/gsc/*.cpp")
add_library(gscvm STATIC ${GSCVM_SOURCES})
target_include_directories(gscvm
    PUBLIC
        "${CMAKE_CURRENT_SOURCE_DIR}/include"
    PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/src"
)
target_link_libraries(gscvm PUBLIC xsk_arc xsk_utils)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(gscvm PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(gscvm PRIVATE -fpermissive)
endif()

# ---------------------------------------------------------------------------
# Command line applications
# ---------------------------------------------------------------------------
if(NOT GSCVM_BUILD_WASM)
    add_executable(gscvm_cli "${CMAKE_CURRENT_SOURCE_DIR}/tools/gscvm.cpp")
    set_target_properties(gscvm_cli PROPERTIES OUTPUT_NAME gscvm)
    target_link_libraries(gscvm_cli PRIVATE gscvm)
endif()

# ---------------------------------------------------------------------------
# Browser WebAssembly module (Emscripten)
# ---------------------------------------------------------------------------
if(GSCVM_BUILD_WASM)
    add_executable(gscvm_wasm "${CMAKE_CURRENT_SOURCE_DIR}/tools/wasm.cpp")
    set_target_properties(gscvm_wasm PROPERTIES
        OUTPUT_NAME "gscvm"
        SUFFIX ".js"
    )
    target_link_libraries(gscvm_wasm PRIVATE gscvm)

    set(GSCVM_WASM_EXPORTS
        _gscvm_init
        _gscvm_shutdown
        _gscvm_set_real_time
        _gscvm_set_tick_rate
        _gscvm_set_execution_mode
        _gscvm_set_instruction_limit
        _gscvm_add_search_path
        _gscvm_clear_search_paths
        _gscvm_add_file
        _gscvm_add_source
        _gscvm_remove_file
        _gscvm_clear_files
        _gscvm_list_files
        _gscvm_resolve_path
        _gscvm_load_file
        _gscvm_load_source
        _gscvm_load_module
        _gscvm_load_dependencies
        _gscvm_call
        _gscvm_start
        _gscvm_tick
        _gscvm_run_until_idle
        _gscvm_run_until_complete
        _gscvm_notify
        _gscvm_threads
        _gscvm_thread_info
        _gscvm_globals
        _gscvm_memory_stats
        _gscvm_memory_blocks
        _gscvm_modules
        _gscvm_disassembly
        _gscvm_module_bytes
        _gscvm_set_breakpoint
        _gscvm_clear_breakpoints
        _gscvm_breakpoints
        _gscvm_debug_state
        _gscvm_step
        _gscvm_resume
        _gscvm_pause
        _gscvm_take_output
        _gscvm_last_error
        _gscvm_free
        _malloc

    )
    string(JOIN "," GSCVM_WASM_EXPORTS_CSV ${GSCVM_WASM_EXPORTS})

    target_link_options(gscvm_wasm PRIVATE
        "-sMODULARIZE=1"
        "-sEXPORT_NAME=createGscvmModule"
        "-sENVIRONMENT=web,worker,node"
        "-sALLOW_MEMORY_GROWTH=1"
        "-sINITIAL_MEMORY=67108864"

        "-sNO_EXIT_RUNTIME=1"
        "-sDISABLE_EXCEPTION_CATCHING=0"
        "-sEXPORTED_FUNCTIONS=${GSCVM_WASM_EXPORTS_CSV}"
        "-sEXPORTED_RUNTIME_METHODS=ccall,cwrap,UTF8ToString,stringToUTF8,lengthBytesUTF8,HEAPU8"
        "-sASSERTIONS=0"

endif()

if(GSCVM_BUILD_UPSTREAM_CLI)
    add_executable(gsc_tool_upstream "${UPSTREAM_DIR}/src/tool/main.cpp")
    set_target_properties(gsc_tool_upstream PROPERTIES OUTPUT_NAME gsc-tool)
    target_include_directories(gsc_tool_upstream PRIVATE "${UPSTREAM_DIR}/deps/cxxopts/include")
    target_link_libraries(gsc_tool_upstream PRIVATE xsk_gsc xsk_arc xsk_utils)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_compile_options(gsc_tool_upstream PRIVATE -fpermissive)
    endif()
endif()

# ---------------------------------------------------------------------------
# Tests (upstream suite + gscvm suite share one Catch2 binary)
# ---------------------------------------------------------------------------
if(GSCVM_BUILD_TESTS OR GSCVM_BUILD_UPSTREAM_TESTS)
    if(NOT EXISTS "${UPSTREAM_DIR}/deps/catch2/extras/catch_amalgamated.cpp")
        message(FATAL_ERROR "Catch2 submodule missing. Run: git -C upstream submodule update --init --recursive")
    endif()

    add_library(xsk_catch2 STATIC
        "${UPSTREAM_DIR}/deps/catch2/extras/catch_amalgamated.cpp"
    )
    target_include_directories(xsk_catch2 PUBLIC "${UPSTREAM_DIR}/deps/catch2/extras")
    target_compile_definitions(xsk_catch2 PUBLIC CATCH_AMALGAMATED_CUSTOM_MAIN)
    if(MSVC)
        target_compile_options(xsk_catch2 PRIVATE /w)
    else()
        target_compile_options(xsk_catch2 PRIVATE -w)
    endif()

    set(GSCVM_TEST_SOURCES)
    if(GSCVM_BUILD_UPSTREAM_TESTS)
        file(GLOB XSK_TEST_SOURCES CONFIGURE_DEPENDS "${UPSTREAM_DIR}/test/*.cpp")
        list(FILTER XSK_TEST_SOURCES EXCLUDE REGEX "/main\\.cpp$")
        list(APPEND GSCVM_TEST_SOURCES ${XSK_TEST_SOURCES})
    endif()
    if(GSCVM_BUILD_TESTS)
        file(GLOB_RECURSE GSCVM_OWN_TEST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/test/*.cpp")
        list(APPEND GSCVM_TEST_SOURCES ${GSCVM_OWN_TEST_SOURCES})
    endif()

    add_executable(gscvm_tests ${GSCVM_TEST_SOURCES})
    set_target_properties(gscvm_tests PROPERTIES OUTPUT_NAME gscvm-tests)
    target_include_directories(gscvm_tests PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/include"
        "${CMAKE_CURRENT_SOURCE_DIR}/test"
    )
    target_compile_definitions(gscvm_tests PRIVATE
        XSK_TEST_DATA="${UPSTREAM_DIR}/test/data"
        GSCVM_TEST_DATA="${CMAKE_CURRENT_SOURCE_DIR}/test/data"
    )
    target_link_libraries(gscvm_tests PRIVATE gscvm xsk_gsc xsk_arc xsk_utils xsk_catch2)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_compile_options(gscvm_tests PRIVATE -fpermissive)
    endif()
    enable_testing()
    add_test(NAME gscvm-tests COMMAND gscvm_tests)
    # The upstream decompiler/recompiler corpus is intentionally large.
    set_tests_properties(gscvm-tests PROPERTIES TIMEOUT 1800)

    add_test(NAME gscvm-cli-source
        COMMAND gscvm_cli "${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures/t6/basic.gsc")
    set_tests_properties(gscvm-cli-source PROPERTIES
        TIMEOUT 60
        PASS_REGULAR_EXPRESSION "main: 7")

    add_test(NAME gscvm-cli-threads
        COMMAND gscvm_cli "${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures/t6/threads.gsc" thread_main)
    set_tests_properties(gscvm-cli-threads PROPERTIES
        TIMEOUT 60
        PASS_REGULAR_EXPRESSION "thread_main: 42")
endif()

install(TARGETS gscvm
    ARCHIVE DESTINATION lib
    RUNTIME DESTINATION bin
)
if(TARGET gscvm_cli)
    install(TARGETS gscvm_cli
        ARCHIVE DESTINATION lib
        RUNTIME DESTINATION bin
    )
endif()
